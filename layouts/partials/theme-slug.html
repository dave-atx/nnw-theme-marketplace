Names are untrusted, so both go through urlize. */}} {{- $slug := .name | urlize }} {{- $shared := where hugo.Data.themes.themes "name" .name }} {{- if or (not $slug) (gt (len $shared) 1) }} {{- $slug = .id | urlize }} {{- end }} {{- return $slug }}
